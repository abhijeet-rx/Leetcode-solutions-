class Solution {
    public void reverseQueue(Queue<Integer> q) {
       Stack<Integer> st = new Stack<>();
       while(!q.isEmpty()){
           int val = q.peek();
           q.remove();
           st.push(val);
       }
       while(!st.isEmpty()){
           int val = st.peek();
           st.pop();
          q.add(val);
       }
       
        
    }
}
