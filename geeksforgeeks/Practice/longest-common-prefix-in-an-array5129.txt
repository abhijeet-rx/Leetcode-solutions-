class Solution {
    public String longestCommonPrefix(String arr[]) {
        // code here
        String ans = "";
        int n = arr.length;
        for(int i =0;i<arr[0].length();i++){
            char ch = arr[0].charAt(i);
            
            boolean match = true;
            for(int j=1;j<n;j++){
                if(arr[j].length()<=i || ch!=arr[j].charAt(i)){
                    match= false;
                    break;
                }
                
            }
            if(match==false){
                break;
            }
            else{
                ans = ans+ch;
            }
        }
        return ans;
    }
}
