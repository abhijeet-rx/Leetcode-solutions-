class Solution {
    public boolean binarySearch(int[] arr, int k) {
        int start = 0;
        int end = arr.length-1;
        int mid = start+(end-start)/2;
        while(start<=end){
            if(arr[mid]>k){
                end = mid-1;
            }
            if(arr[mid]<k){
                start = mid+1;
            }
            if(arr[mid]==k){
                return true;
            }
            mid = start+(end-start)/2;
        }
            return false;
        
    }
}
