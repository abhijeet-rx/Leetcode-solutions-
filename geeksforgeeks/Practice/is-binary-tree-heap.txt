class Solution {
    int findcount(Node root){
        if(root==null){
            return 0;
        }
        int ans  = findcount(root.left)+findcount(root.right)+1;
        return ans;
    }
    boolean cbt(Node root , int index  , int count){
        if(root==null){
            return true;
        }
        if(index>=count){
            return false;
        }
        else{
        boolean left =  cbt(root.left, index*2+1,count);
        boolean right =  cbt(root.right, index*2+2,count);
        return (left && right);
             
          }   
        }   
        boolean maxheap(Node root){
            if(root.left==null && root.right == null){
                return true;
            }
            if(root.right==null){
                return (root.data>root.left.data && maxheap(root.left));
            }
            else{
                boolean left = maxheap(root.left);
                boolean right = maxheap(root.right);
                
            return (left && right && root.data> root.left.data && root.data > root.right.data);
            }
        }
    
    public boolean isHeap(Node tree) {
      int index = 0;
      int count = findcount(tree);
      if(cbt(tree,index,count) && maxheap(tree)){
          return true;
      }
      else{
          return false;
      }
    }
}
