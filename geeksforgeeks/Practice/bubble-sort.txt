class Solution {
    public void bubbleSort(int[] arr) {
       int n = arr.length;
       for(int i=1;i<n;i++){
           for(int j =0;j<n-1;j++)
           if(arr[j]>arr[i]){
               int temp = arr[i];
               arr[i] = arr[j];
               arr[j] = temp;
           }
       }
        
    }
}
