/* Definition for Node
class Node {
    public int data;
    public Node left;
    public Node right;

    public Node(int val) {
        data = val;
        left = right = null;
    }
};
*/

class Solution {
    int k;
    boolean found;
    Node solve(Node root,  int node){
        if(root==null){
            return null;
        }
        if(root.data==node){
            return root;
        }
        Node left = solve(root.left,node);
            if(left!=null ){
                if(found){
                    return left;
                }
                if(k>0){
                    k--;
                }
                if(k==0){
                    found=true;
                    return root;
                }
                return left;
            }
        Node right = solve(root.right,node);
            if( right!=null){
                if(found){
                    return right;
                }
                if(k>0){
                    k--;
                }
                if(k==0){
                    found=true;
                    return root;
                }
                return right;
            }
           return null;
    }
    public int kthAncestor(Node root, int k, int node) {
        this.k=k;
        this.found=false;
        // code here
      Node ans = solve(root,node);
      if(found==false){
          return -1;
        
    }
    else{
        return ans.data;
    }
}
}
