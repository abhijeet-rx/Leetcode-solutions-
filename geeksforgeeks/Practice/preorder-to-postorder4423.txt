class Solution {
    Node insert(int value ,Node root){
        if(root==null){
            return new Node(value);
        }
        if(value>root.data){
            root.right = insert(value,root.right);
        }
        else{
            root.left = insert(value, root.left);
        }
        return root;
    }
    public Node preToBST(int pre[]) {
        Node root = null;
        for(int val:pre){
            root = insert(val , root);
        }
        return root;
        
        
    }
}
