class Solution {
      
       int maxSum = Integer.MIN_VALUE;;
       int maxLen = 0;
    void  solve(Node root, int len, int sum){
        if(root==null){
            if(len>maxLen){
                 maxLen = len;
                 maxSum = sum;
            }
            if(len==maxLen){
            maxSum  = Math.max(sum,maxSum);
                
            }
            return;
        }
        sum = sum+root.data;
        solve(root.left,len+1,sum);
        solve(root.right,len+1,sum);
     
    }
    public int sumOfLongRootToLeafPath(Node root) {
       solve(root,0,0);
       return maxSum;
        
    }
}
