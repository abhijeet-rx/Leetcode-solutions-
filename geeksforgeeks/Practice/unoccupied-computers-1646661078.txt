class Solution {
    public int solve(int n, String s) {
        int ans = 0;

        Stack<Character> st = new Stack<>();
        HashSet<Character> rejected = new HashSet<>();

        for (int i = 0; i < s.length(); i++) {
            char ch = s.charAt(i);

            
            if (!st.contains(ch) && !rejected.contains(ch)) {

                if (st.size() < n) {
                    st.push(ch);
                } else {
                    rejected.add(ch);
                    ans++;
                }

            } 
            
            else if (st.contains(ch)) {
                st.remove((Character) ch);
            }
        }

        return ans;
    }
}
