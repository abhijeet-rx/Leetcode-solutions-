class Solution {
    public Queue<Integer> reverseFirstK(Queue<Integer> q, int k) {
        int n = q.size();
        if(k>n) return q;
        Stack<Integer> st = new Stack<>();
        for(int i =0;i<k;i++){
            int val = q.remove();
            st.push(val);
        }
        for(int i=0;i<k;i++){
            q.add(st.peek());
            st.pop();
        }
        for(int i=0;i<n-k;i++){
            int val =q.remove();
            q.add(val);
        }
        return q;
        
        
    }
}
