class Solution {
    public void rotateMatrix(int[][] mat) {
        int n = mat.length;
        for(int i = 0;i<n;i++){
            for(int j =i+1;j<n;j++){
                int temp  = mat[i][j];
                mat[i][j] = mat[j][i];
                mat[j][i] = temp;
            }
        }
        for(int i=0;i<n;i++){
            int start = 0;
        int  end = n-1;
        while(start<end){
            int temp  = mat[start][i];
            mat[start][i] = mat[end][i];
            mat[end][i] = temp;
            start++;
            end--;
        }
        }
        
    }
}
