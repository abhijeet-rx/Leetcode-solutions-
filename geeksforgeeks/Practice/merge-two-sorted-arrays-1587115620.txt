class Solution {
    public void mergeArrays(int a[], int b[]) {
        int n = a.length;
        int m = b.length;
        int z = n+m;
        int  i =0;
        int j = 0;
        int k =0;
        int[] ans = new int[z];
        while(i<n && j<m){
            if(a[i]<b[j]){
                ans[k++] = a[i++];
        }
        else{
            ans[k++] = b[j++];
        }
        
        
    }
    while(i<n){
        ans[k++] = a[i++];
    }
    while(j<m){
        ans[k++] = b[j++];
    }
    for(int x=0;x<n;x++){
        a[x] = ans[x];
    }
    for(int x=0;x<m;x++){
        b[x] = ans[n+x];
    }
    
}
}
