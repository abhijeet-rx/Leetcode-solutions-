/*
Definition for Node
class Node {
    int data;
    Node left, right;

    public Node(int val)
    {
        data = val;
        left = right = null;
    }
}
*/

class Solution {
    int i =0;
    int solve(Node root  , int k){
        if(root==null){
            return -1;
            
        }
        int left  = solve(root.left  , k);
        if(left!=-1){
            return left;
        }
        i++;
        if(i==k){
            return root.data;
            
        }
        else
        return solve(root.right,k);
    }
    public int kthSmallest(Node root, int k) {
        int ans  = solve(root ,k);
        return ans;
        
    }
}
