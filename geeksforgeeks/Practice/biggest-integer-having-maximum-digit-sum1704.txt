class Solution {
    public int findMax(int n) {

        int ans = n;
        int maxSum = digitSum(n);

        int place = 1;

        while (place <= n) {

            int candidate = (n / (place * 10)) * (place * 10)
                          + ((n / place) % 10 - 1) * place
                          + (place - 1);

            if (candidate >= 1) {
                int sum = digitSum(candidate);

                if (sum > maxSum || (sum == maxSum && candidate > ans)) {
                    maxSum = sum;
                    ans = candidate;
                }
            }

            place *= 10;
        }

        return ans;
    }

    private int digitSum(int num) {
        int sum = 0;

        while (num > 0) {
            sum += num % 10;
            num /= 10;
        }

        return sum;
    }
}
